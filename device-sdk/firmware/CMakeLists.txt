cmake_minimum_required(VERSION 3.16)
set(CLINE_BOARD "" CACHE STRING "Cline device profile (must be selected explicitly)")
if(NOT CLINE_BOARD)
    message(FATAL_ERROR "No device selected. Run 'bun run device' from the repository root, or pass -D CLINE_BOARD=<board> with -B build/<board>. There is no default board.")
endif()
get_filename_component(build_board "${CMAKE_BINARY_DIR}" NAME)
if(NOT build_board STREQUAL CLINE_BOARD)
    message(FATAL_ERROR "Use a separate board build directory: -B build/${CLINE_BOARD}. Run 'bun run device' to choose a device and serial port.")
endif()
set(profile "${CMAKE_CURRENT_LIST_DIR}/devices/sdkconfig.cline-${CLINE_BOARD}")
if(NOT EXISTS "${profile}")
    message(FATAL_ERROR "Unknown Cline board: ${CLINE_BOARD}")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${profile}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_BINARY_DIR}/dependencies.lock")
project(cline_device)
if(NOT CONFIG_DEVICE_BOARD_ID STREQUAL CLINE_BOARD)
    message(FATAL_ERROR "Build directory is configured for ${CONFIG_DEVICE_BOARD_ID}, not ${CLINE_BOARD}. Use a separate build directory per board.")
endif()
